import React, { useEffect } from 'react'
import { useNavigate } from 'react-router-dom'
import { useSateliteActores } from '../../context/ContextSateliteActores'
import { DataTableCustom } from '../DataTableCustom'
import type { TableColumn } from 'react-data-table-component'
import {
    ContainerLabel,
    ContainerSelectTable,
    EstadoRow,
    EstadoField,
    StickyTableStyles
} from './TableTodos.styled'
import { Row } from '../Row'
import type { Obra } from '../../types/obra'
import type { Pago } from '../../types/pagos'
import apiClient from '../../api/api'
import { fetchPagos } from '../../services/getPagos.service'
import { message } from 'antd'
import dayjs from 'dayjs'
import { usePagination } from '../../hooks/usePagination'

interface EstadoAtencion {
    id: number
    name: string
    color?: string
}

export const TableTodos: React.FC = () => {
    const {
        empresaId: rawEmpresaId,
        obrasFiltradas,
        setKpis
    } = useSateliteActores()
    const {
        currentPage,
        handlePageChange,
        handleRowsPerPageChange,
        rows,
        rowsPage,
        rowsTotal
    } = usePagination<Obra>(obrasFiltradas ?? null)
    const [estadosAtencion, setEstadosAtencion] = React.useState<
        EstadoAtencion[]
    >([])
    const [pagos, setPagos] = React.useState<Pago[]>([])
    const [loading, setLoading] = React.useState(false)

    const empresaId =
        typeof rawEmpresaId === 'string'
            ? parseInt(rawEmpresaId, 10)
            : rawEmpresaId

    if (isNaN(empresaId)) {
        console.error('empresaId is not a valid number:', rawEmpresaId)
    }

    const [seccionesCompletadas, setSeccionesCompletadas] = React.useState<{
        [obraId: number]: { [seccionId: number]: boolean }
    }>({})

    const cargarDatos = async () => {
        setLoading(true)
        try {
            const estadosResponse = await apiClient.get(
                '/all/estadodeatencion',
                {
                    params: { id_empresa: empresaId }
                }
            )
            setEstadosAtencion(estadosResponse.data.data || [])

            const obrasValidas = (obrasFiltradas || [])
                .filter((obra): obra is Obra => {
                    const isValid =
                        obra !== undefined &&
                        typeof obra.id === 'number' &&
                        !isNaN(obra.id)
                    if (!isValid) {
                        console.warn('[cargarDatos] Obra inválida:', obra)
                    }
                    return isValid
                })
                .map((obra) => ({
                    ...obra,
                    centros_operacion: Array.isArray(obra.centros_operacion)
                        ? obra.centros_operacion
                        : []
                }))

            if (obrasValidas.length === 0) {
                setPagos([])
                setLoading(false)
                return
            }
            
        } catch (error: any) {
            console.error('Error al cargar datos:', error)
            message.error('No se pudieron cargar los datos.')
            setEstadosAtencion([])
            setPagos([])
        } finally {
            setLoading(false)
        }
    }

    useEffect(() => {
        cargarDatos()
    }, [empresaId, obrasFiltradas])

    const getSiguienteEstado = (
        currentEstadoId: number,
        obraId: number
    ): number | null => {
        const seccionesObra = seccionesCompletadas[obraId] || {}
        const estadosOrdenados = [...estadosAtencion].sort(
            (a, b) => a.id - b.id
        )
        const currentIndex = estadosOrdenados.findIndex(
            (estado) => estado.id === currentEstadoId
        )

        for (let i = currentIndex + 1; i < estadosOrdenados.length; i++) {
            const nextEstado = estadosOrdenados[i]
            if (!seccionesObra[nextEstado.id]) {
                return nextEstado.id
            }
        }

        return (
            estadosOrdenados[estadosOrdenados.length - 1]?.id || currentEstadoId
        )
    }

    const getMontoPagado = (obra: Obra) => {
        return formatCurrency(obra.monto_pagado || 0)
    }

    const getMontoRecuperado = (obra: Obra) => {
        return formatCurrency(obra.monto_recuperado || 0)
    }

    const getEstadoInfo = (estadoId?: number) => {
        const estado = estadosAtencion.find((e) => e.id === estadoId)
        return {
            name: estado?.name ?? 'N/A',
            color: estado?.color ?? '#000000'
        }
    }

    const formatCurrency = (value: number | string | undefined): string => {
        if (value === undefined || value === null || value === '') {
            return 'S/. 0.00'
        }
        const numericValue =
            typeof value === 'string'
                ? parseFloat(value.replace(/[^0-9.-]+/g, ''))
                : value
        if (isNaN(numericValue)) {
            return 'S/. 0.00'
        }
        const fixedValue = numericValue.toFixed(2)
        const [integerPart, decimalPart] = fixedValue.split('.')
        const formattedInteger = integerPart.replace(
            /\B(?=(\d{3})+(?!\d))/g,
            ','
        )
        return `S/. ${formattedInteger}.${decimalPart}`
    }

    const getCostoProyecto = (obra: Obra) => {
        return formatCurrency(obra.costo_proyecto || 0)
    }

    const formatDate = (date: string | undefined): string => {
        if (!date) return 'N/A'
        return dayjs(date).format('DD/MM/YYYY')
    }

    const columns: TableColumn<Obra>[] = [
        {
            name: 'Obra',
            center: true,
            minWidth: '200px',
            grow: 0,
            cell: (row: Obra) => (
                <Row>
                    <ContainerLabel pointer="true">
                        {row.nombre || 'Sin nombre'}
                    </ContainerLabel>
                </Row>
            )
        },
        {
            name: 'Centro de Operación',
            center: true,
            minWidth: '200px',
            grow: 0,
            cell: (row: Obra) => (
                <div
                    style={{
                        display: 'flex',
                        flexDirection: 'column',
                        alignItems: 'center'
                    }}
                >
                    {Array.isArray(row.centros_operacion) &&
                    row.centros_operacion.length > 0 ? (
                        row.centros_operacion.map((centro, index) => (
                            <span key={index}>{centro.nombre || 'N/A'}</span>
                        ))
                    ) : (
                        <span>N/A</span>
                    )}
                </div>
            )
        },
        {
            name: 'Fecha de Inicio',
            minWidth: '200px',
            center: true,
            grow: 0,
            cell: (row: Obra) => <span>{formatDate(row.fecha_reembolso)}</span>
        },
        {
            name: 'Fecha de Culminación',
            minWidth: '200px',
            center: true,
            grow: 0,
            cell: (row: Obra) => <span>{formatDate(row.fecha_conclusion)}</span>
        },
        {
            name: 'Estado',
            minWidth: '190px',
            center: true,
            width: '200px',
            grow: 1,
            cell: (row: Obra) => {
                const { name, color } = getEstadoInfo(row.estado_id)
                return (
                    <ContainerSelectTable>
                        <EstadoRow>
                            <EstadoField $backgroundColor={color}>
                                {name}
                                <svg
                                    xmlns="http://www.w3.org/2000/svg"
                                    width="20"
                                    height="20"
                                    viewBox="0 0 24 24"
                                    fill="none"
                                    stroke="#ffffff"
                                    strokeWidth="2"
                                    strokeLinecap="round"
                                    strokeLinejoin="round"
                                >
                                    <path d="M18 13v6a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2V8a2 2 0 0 1 2-2h6"></path>
                                    <polyline points="15 3 21 3 21 9"></polyline>
                                    <line x1="10" y1="14" x2="21" y2="3"></line>
                                </svg>
                            </EstadoField>
                        </EstadoRow>
                    </ContainerSelectTable>
                )
            }
        },
        {
            name: 'Costo del Proyecto',
            minWidth: '200px',
            center: true,
            grow: 0,
            cell: (row: Obra) => <span>{getCostoProyecto(row)}</span>
        },
        {
            name: 'Monto Pagado',
            minWidth: '200px',
            center: true,
            grow: 0,
            cell: (row: Obra) => <span>{getMontoPagado(row)}</span>
        },
        {
            name: 'Monto Recuperado',
            minWidth: '200px',
            center: true,
            grow: 0,
            cell: (row: Obra) => <span>{getMontoRecuperado(row)}</span>
        }
    ]

    const navigate = useNavigate()
    
    const handleOnViewComponent = (row: Obra) => {
        navigate(`/detalles/${row.id}`)
    }

    const obrasValidas = (obrasFiltradas || [])
        .filter((obra): obra is Obra => {
            const isValid =
                obra !== undefined &&
                typeof obra.id === 'number' &&
                !isNaN(obra.id)
            if (!isValid) {
                console.warn('[TableTodos] Obra inválida:', obra)
            }
            return isValid
        })
        .map((obra) => ({
            ...obra,
            centros_operacion: Array.isArray(obra.centros_operacion)
                ? obra.centros_operacion
                : []
        }))

    return (
        <div className="row" style={{ position: 'relative' }}>
            <div
                className="col-md-12 md-max-type-b"
                style={{ backgroundColor: 'white', marginTop: 20 }}
            >
                {<StickyTableStyles />}
                <DataTableCustom
                    title=""
                    columns={columns}
                    data={rows || []}
                    totalRows={rowsTotal}
                    currentPage={currentPage}
                    rowsPerPage={rowsPage}
                    onPageChange={handlePageChange}
                    onRowsPerPageChange={handleRowsPerPageChange}
                    emptyText="No existen Obras registradas"
                    stickyColumns
                    onViewComponent={handleOnViewComponent}
                />
            </div>
        </div>
    )
}

export default TableTodos
